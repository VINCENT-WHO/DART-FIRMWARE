cmake_minimum_required(VERSION 3.21)
project(DartGreenLightDetector VERSION 1.0 LANGUAGES CXX)

# 独立电脑端工程：使用本机C++编译器，不使用STM32工具链。
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)

add_library(green_detector src/detector.cpp)
target_include_directories(green_detector PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(green_detector PUBLIC ${OpenCV_LIBS})
add_executable(green_light_detector src/main.cpp)
target_link_libraries(green_light_detector PRIVATE green_detector)
if(WIN32)
    target_link_libraries(green_light_detector PRIVATE shell32)
endif()
foreach(target green_detector green_light_detector)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(detector_tests tests/detector_tests.cpp)
    target_link_libraries(detector_tests PRIVATE green_detector)
    add_test(NAME detector_synthetic COMMAND detector_tests)
endif()
